# meknik-ui Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A GTK 4 window, `meknik-ui`, that shows and sets the meknik daemon's volume and mute, built and tested in a pinned podman build image.

**Architecture:** First move every build, test, and lint into a Debian 13.7 podman image. Then extract the code that both binaries need (state paths, daemon spawn, "ask the daemon, starting it if needed, and parse the reply") out of `cmd/meknik` into GTK-free packages with unit tests. Last, add `cmd/meknik-ui`, a thin gotk4 layer over those packages, checked by hand.

**Tech Stack:** Go 1.26 (Debian trixie-backports `golang-1.26-go`), gotk4 v0.4.1 (`github.com/diamondburned/gotk4/pkg`), GTK 4.18, podman 5.4, golangci-lint v2.12.2, Fossil.

**Spec:** `docs/superpowers/specs/2026-10-07-meknik-ui-design.md`

## Global Constraints

- Base image `debian:13.7`; GTK from trixie (`libgtk-4-dev`); Go from trixie-backports (`golang-1.26-go`); `GOTOOLCHAIN=local`.
- golangci-lint v2.12.2 via `go install github.com/golangci/golangci-lint/v2/cmd/golangci-lint@v2.12.2`.
- `mise.toml` is deleted. `go.mod` keeps `go 1.26.7`.
- `bin/meknik` is built with `CGO_ENABLED=0`; `bin/meknik-ui` with cgo.
- Only `meknik` goes to `~/.local/share/meknik/container-bin`.
- Application ID `tel.lil.meknik`; window title `meknik`.
- Volume scale: `minVolumeDB` −60 dB to `control.MaxVolumeDB` (+24 dB), 0.5 dB steps; `volumeDebounce` 100 ms.
- The mute button sends `mute` or `unmute`, never `toggle`.
- Setting a control from a reply never sends a request.
- The daemon and the control protocol do not change.
- Test invocation: `make test`. Lint: `make lint`. Both run in the image and must pass before every commit.
- Commits go to Fossil, one line, no trailers, listing each path: `fossil commit -m "<msg>" path ...`. New files need `fossil add` first. Never `fossil revert`.
- Code style (`~/.claude/CLAUDE.md`): no comments beyond one-line doc comments and package headers; one assertion per test; tests under 5 s; named constants; no defensive parameter checks; no fallbacks; `slog` for logging.

## Review Focus

1. **Daemon volume below −60 dB** (set with `meknik volume -80`): the scale shows −60, and opening the window must not send a request that changes the daemon to −60. Checked by hand in Task 6, Step 8.
2. **No daemon running when the window opens**: the window must start one and show its settings. Pinned by `TestControlSpawnsDaemonWhenNoneAnswers` in Task 5, and checked by hand in Task 6.
3. **Daemon refuses a request** (for example, `mixer.json` cannot be saved): the error must reach the caller as `transport.ErrDaemon`, not as a parse failure. Pinned by `TestControlReportsDaemonError` in Task 5.
4. **Version skew** between an installed daemon and the window, so the status line differs: `ParseSettings` must reject it with `control.ErrStatus`, not invent settings. Pinned by `TestParseSettingsRejectsMalformedLines` in Task 4.
5. **Replies arrive out of order** (a slow volume reply lands after a later mute reply): the window can briefly show stale state. This is accepted for a local socket with millisecond replies. The reviewer should confirm that each reply sets both controls, so the next reply corrects it.

```go
// Package statedir locates meknik's state files and sockets.
package statedir

import (
	"os"
	"path/filepath"
)

// ControlSocket is the name of the daemon's control socket in the run directory.
const ControlSocket = "meknik.ctl"

const (
	appName    = "meknik"
	runDirName = "run"
	logName    = "meknik.log"
	logMode    = 0o644
	dirMode    = 0o755
)

// Path returns the path of a file in meknik's state directory.
func Path(name string) (string, error) {
	dir, err := stateDir()
	return filepath.Join(dir, name), err
}

// RunPath returns the path of a file in the run directory, which containers share.
func RunPath(name string) (string, error) { return Path(filepath.Join(runDirName, name)) }

// OpenLog opens the shared log file for appending, creating the state directory.
func OpenLog() (*os.File, error) {
	path, err := Path(logName)
	if err != nil {
		return nil, err
	}
	if err := os.MkdirAll(filepath.Dir(path), dirMode); err != nil {
		return nil, err
	}
	return os.OpenFile(path, os.O_CREATE|os.O_APPEND|os.O_WRONLY, logMode)
}

func stateDir() (string, error) {
	if dir := os.Getenv("XDG_STATE_HOME"); dir != "" {
		return filepath.Join(dir, appName), nil
	}
	home, err := os.UserHomeDir()
	return filepath.Join(home, ".local", "state", appName), err
}
```

package daemon

import (
	"errors"
	"os"
	"os/exec"
	"syscall"

	"meknik/statedir"
)

// Command returns the detached `meknik daemon` process, logging its stderr to logFile.
func Command(binary string, logFile *os.File) *exec.Cmd {
	cmd := exec.Command(binary, "daemon")
	cmd.Dir = "/"
	cmd.Stderr = logFile
	cmd.SysProcAttr = &syscall.SysProcAttr{Setsid: true}
	return cmd
}

// Spawn starts a detached daemon from binary.
func Spawn(binary string) error {
	logFile, err := statedir.OpenLog()
	if err != nil {
		return err
	}
	cmd := Command(binary, logFile)
	if err := cmd.Start(); err != nil {
		return errors.Join(err, logFile.Close())
	}
	return errors.Join(cmd.Process.Release(), logFile.Close())
}
```

const (
	mutedState   = "muted"
	unmutedState = "unmuted"
	statusFormat = "volume %g dB, %s"
)

// ErrStatus reports a status line that ParseSettings cannot read.
var ErrStatus = errors.New("invalid status line")

// String reports the settings as one status line.
func (s Settings) String() string {
	state := unmutedState
	if s.Muted {
		state = mutedState
	}
	return fmt.Sprintf(statusFormat, s.VolumeDB, state)
}

// ParseSettings reads a status line written by Settings.String.
func ParseSettings(line string) (Settings, error) {
	var s Settings
	var state string
	_, err := fmt.Sscanf(line, statusFormat, &s.VolumeDB, &state)
	if err != nil || !slices.Contains([]string{mutedState, unmutedState}, state) {
		return Settings{}, fmt.Errorf("%w: %q", ErrStatus, line)
	}
	s.Muted = state == mutedState
	return s, nil
}
```

```go
const (
	// SpawnTimeout bounds how long a client waits for a daemon it started.
	SpawnTimeout = time.Second
	// RetryInterval is the pause between attempts to reach a starting daemon.
	RetryInterval = 20 * time.Millisecond
)

// Control sends a request, starting a daemon with spawn when none answers, and returns the resulting settings.
func Control(path string, r control.Request, spawn func() error) (control.Settings, error) {
	var status string
	ask := func() (err error) {
		status, err = Ask(path, r)
		return err
	}
	if err := Deliver(ask, spawn, SpawnTimeout, RetryInterval); err != nil {
		return control.Settings{}, err
	}
	return control.ParseSettings(status)
}
```

```go
// Command meknik-ui is a GTK 4 window that sets the meknik daemon's volume and mute.
package main

import (
	"fmt"
	"log/slog"
	"os"
	"path/filepath"
	"time"

	"github.com/diamondburned/gotk4/pkg/core/glib"
	"github.com/diamondburned/gotk4/pkg/gio/v2"
	"github.com/diamondburned/gotk4/pkg/gtk/v4"

	"meknik/control"
	"meknik/daemon"
	"meknik/statedir"
	"meknik/transport"
)

const (
	appID          = "tel.lil.meknik"
	title          = "meknik"
	daemonBinary   = "meknik"
	muteLabel      = "Mute"
	minVolumeDB    = -60.0
	volumeStepDB   = 0.5
	volumeDigits   = 1
	volumeDebounce = 100 * time.Millisecond
	spacing        = 6
)

type panel struct {
	volume  *gtk.Scale
	mute    *gtk.ToggleButton
	errors  *gtk.Label
	ask     func(control.Request) (control.Settings, error)
	pending glib.SourceHandle
	syncing bool
}

func main() {
	logFile, err := statedir.OpenLog()
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	slog.SetDefault(slog.New(slog.NewTextHandler(logFile, nil)))
	app := gtk.NewApplication(appID, gio.ApplicationDefaultFlags)
	app.ConnectActivate(func() { newPanel(app, askDaemon).send(control.Request{Command: control.Status}) })
	os.Exit(app.Run(os.Args))
}

func askDaemon(r control.Request) (control.Settings, error) {
	path, err := statedir.RunPath(statedir.ControlSocket)
	if err != nil {
		return control.Settings{}, err
	}
	return transport.Control(path, r, spawnSibling)
}

func spawnSibling() error {
	self, err := os.Executable()
	if err != nil {
		return err
	}
	return daemon.Spawn(filepath.Join(filepath.Dir(self), daemonBinary))
}

func newPanel(app *gtk.Application, ask func(control.Request) (control.Settings, error)) *panel {
	p := &panel{
		volume: gtk.NewScaleWithRange(gtk.OrientationHorizontal, minVolumeDB, control.MaxVolumeDB, volumeStepDB),
		mute:   gtk.NewToggleButtonWithLabel(muteLabel),
		errors: gtk.NewLabel(""),
		ask:    ask,
	}
	p.volume.SetDrawValue(true)
	p.volume.SetDigits(volumeDigits)
	p.volume.SetHExpand(true)
	p.volume.ConnectValueChanged(p.volumeChanged)
	p.mute.ConnectToggled(p.muteToggled)
	box := gtk.NewBox(gtk.OrientationVertical, spacing)
	box.Append(p.volume)
	box.Append(p.mute)
	box.Append(p.errors)
	window := gtk.NewApplicationWindow(app)
	window.SetTitle(title)
	window.SetChild(box)
	window.SetVisible(true)
	return p
}

func (p *panel) volumeChanged() {
	if p.syncing {
		return
	}
	if p.pending != 0 {
		glib.SourceRemove(p.pending)
	}
	p.pending = glib.TimeoutAdd(uint(volumeDebounce.Milliseconds()), func() {
		p.pending = 0
		p.send(control.Request{Command: control.Volume, VolumeDB: p.volume.Value()})
	})
}

func (p *panel) muteToggled() {
	if p.syncing {
		return
	}
	command := control.Unmute
	if p.mute.Active() {
		command = control.Mute
	}
	p.send(control.Request{Command: command})
}

func (p *panel) send(r control.Request) {
	go func() {
		settings, err := p.ask(r)
		glib.IdleAdd(func() { p.show(settings, err) })
	}()
}

func (p *panel) show(settings control.Settings, err error) {
	if err != nil {
		slog.Error("control request failed", "err", err)
		p.errors.SetText(err.Error())
		return
	}
	p.errors.SetText("")
	p.syncing = true
	p.volume.SetValue(settings.VolumeDB)
	p.mute.SetActive(settings.Muted)
	p.syncing = false
}
```
